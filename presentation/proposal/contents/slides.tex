    \begin{itemize}

% B4: Power Analysis Detail
\begin{frame}
    \frametitle{Backup: Analisis \textit{Power} dan Justifikasi Ukuran Sampel}
    \begin{itemize}
        \item \textbf{Ukuran efek target:} $d = 1{,}0$ (Cohen's convention, efek besar), berdasarkan temuan Kaggantinataraja (2025)
        \item \textbf{Power dan signifikansi:} $1 - \beta = 0{,}80$, $\alpha = 0{,}05$ (\textit{two-tailed})
        \item \textbf{Ukuran sampel minimum:} $n = 17$ secara total per metrik (Cohen, 1988)
        \item \textbf{R2:} $7 \times 10 = 70$ observasi per lingkungan $\gg 17$
        \item \textbf{R1:} $7 \times 5 = 35$ observasi per lingkungan, memadai untuk efek besar
        \item \textbf{R3:} Deskriptif, tidak memerlukan ukuran sampel statistik
    \end{itemize}
\end{frame}
