% =============================================
% 3. Configuration drift: definition and evidence
% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Configuration Drift: Definisi dan Bukti}
    \vspace{0.05cm}
    \begin{columns}[T]
        \begin{column}{0.46\textwidth}
            \textbf{Definisi} {\small (Pohjola, 2025)}

            \smallskip
            State aktual klaster tidak lagi sesuai dengan state yang dideklarasikan.

            \vspace{0.20cm}
            \textbf{Penyebab umum}
            \begin{enumerate}
                \item Perubahan langsung tanpa \textit{source of truth}
                \item Prosedur deployment tidak konsisten
                \item Tidak ada deteksi otomatis
            \end{enumerate}
        \end{column}
        \begin{column}{0.50\textwidth}
            \textbf{Bukti empiris}
            \begin{itemize}
                \item 71\% dari 2.260 skrip K8s mengandung \textit{misconfiguration} bermakna {\small (Zhang et al., 2026)}.
                \item 11 kategori \textit{security misconfiguration} berulang pada manifest K8s {\small (Rahman et al., 2023)}.
                \item Drift berdampak pada keandalan, keamanan, dan proses pemulihan layanan.
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.08cm}
    \slidelead{Karena itu, drift perlu diukur sebagai fenomena operasional, bukan hanya dijelaskan sebagai konsep.}
\note{
Pembuka: berikan definisi singkat agar semua penguji memakai istilah yang sama.

Pesan utama: drift bukan masalah abstrak; literatur menunjukkan misconfiguration Kubernetes sering terjadi dan berdampak nyata.

Transisi: setelah masalahnya terlihat, jelaskan respons yang umum diajukan, yaitu GitOps.
}
\end{frame}
}

                \item Agen di klaster membandingkan Git dengan state aktual.

% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Rumusan Masalah}
    \vspace{0.08cm}
    \begin{beamercolorbox}[wd=\textwidth, sep=0.25cm, center, rounded=true]{structure}
        \normalsize\bfseries
        Sejauh mana deployment manual dengan \texttt{kubectl} dan GitOps dengan ArgoCD menghasilkan respons operasional yang berbeda ketika menghadapi \textit{configuration drift} terkontrol?
    \end{beamercolorbox}
    \vspace{0.22cm}
    \begin{enumerate}
        \item \textbf{R1 -- Konsistensi:} apakah deployment kembali ke state yang diharapkan?
        \item \textbf{R2 -- Waktu:} berapa lama drift dideteksi, dipulihkan, dan diidentifikasi?
        \item \textbf{R3 -- Traceability:} seberapa lengkap jejak perubahan dapat direkonstruksi?
    \end{enumerate}
\note{
Pembuka: bacakan pertanyaan utama secara pelan karena ini pusat proposal.

Pesan utama: ada tiga dimensi yang diukur: konsistensi, waktu respons, dan traceability.

Transisi: setelah pertanyaan jelas, batasi ruang lingkup agar klaim penelitian tidak terlalu luas.
}

% 15. Statistical analysis
% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Analisis Statistik}
    \vspace{0.05cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Metode analisis}
            \begin{itemize}
                \item Statistik deskriptif per metrik.
                \item \textit{Mann--Whitney U}, $\alpha = 0{,}05$.
                \item Ukuran efek: Cohen's $d$ atau Cliff's $\delta$.
                \item R3 dianalisis secara deskriptif.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Justifikasi}
            \begin{itemize}
                \item \textit{Mann--Whitney}: data waktu belum tentu normal dan sampel per skenario relatif kecil.
                \item $n = 10$ per skenario: total $7 \times 10 = 70$ observasi per lingkungan untuk R2.
            \end{itemize}
        \end{column}
    \end{columns}
\note{
Pembuka: jelaskan bahwa analisis dipilih sebelum pengumpulan data.

Pesan utama: uji non-parametrik dipilih karena data waktu sistem dapat tidak normal; ukuran efek tetap dilaporkan agar hasil punya makna praktis.

Transisi: setelah metode analisis, jelaskan bahwa semua kemungkinan hasil tetap informatif.
}
\end{frame}

% B5: Statistical analysis detail

\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Backup: Analisis Statistik dan Ukuran Sampel}
    \vspace{0.02cm}
    \scriptsize
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Analisis}
            \begin{itemize}
                \item Statistik deskriptif per metrik.
                \item Mann--Whitney U, $\alpha = 0{,}05$.
                \item Ukuran efek: Cohen's $d$ atau Cliff's $\delta$.
                \item R3 dianalisis secara deskriptif.
            \end{itemize}
            \vspace{0.10cm}
            \textbf{Mengapa Mann--Whitney?}
            \begin{itemize}
                \item Data waktu sistem belum tentu normal.
                \item Sampel per skenario relatif kecil.
                \item Uji non-parametrik tidak mensyaratkan normalitas.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Justifikasi ukuran sampel}
            \begin{itemize}
                \item Power = 0.80, $\alpha = 0{,}05$, target efek besar ($d = 1{,}0$).
                \item Minimum $n = 17$ per grup secara total.
                \item R2: $7 \times 10 = 70$ observasi per lingkungan.
                \item R1: $7 \times 5 = 35$ observasi per lingkungan.
                \item R3: deskriptif, tidak memerlukan ukuran sampel statistik.
            \end{itemize}
            \vspace{0.06cm}
            \textbf{Mengapa $n = 10$ per skenario?}
            \begin{itemize}
                \item Studi sistem berulang dan layak dalam batasan eksperimen.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

{\begin{itemize}
    \item Mulai dari contoh perubahan cepat langsung di klaster.
    \item Jelaskan bahwa masalah muncul ketika dokumentasi tidak ikut berubah.

    \item Sebutkan tiga penyebab umum.
    \item Gunakan angka 71\% sebagai bukti bahwa misconfiguration bukan kasus langka.
\end{itemize}}
{\begin{itemize}
    \item Jelaskan bahwa GitOps hadir sebagai respons terhadap masalah ini.
\end{itemize}}

{GitOps menjanjikan reconciliation otomatis antara Git dan state klaster.}

\speakerpoints{Analisis Statistik}
