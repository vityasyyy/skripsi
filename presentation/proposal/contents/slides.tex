% =============================================
% 2. What is application deployment?
% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Apa Itu Deployment Aplikasi?}
    \begin{columns}[T]
        \begin{column}{0.44\textwidth}
            \scriptsize
            \textbf{Deployment} adalah proses membawa aplikasi dari kode menjadi layanan yang berjalan: membangun, mengemas, mengirim, dan menjalankan di klaster.
            \vspace{0.04cm}
            \textbf{Aplikasi multi-service} memiliki banyak komponen yang saling bergantung. InvenioRDM, misalnya, terdiri dari 16+ komponen: web, worker, database, search, cache, object storage, ingress, secrets, dan monitoring.
            \vspace{0.04cm}
            \textbf{Dua paradigma deployment:}
            \begin{itemize}\setlength{\itemsep}{0pt}
                \item \textbf{Manual:} operator menjalankan \texttt{kubectl} langsung ke klaster.
                \item \textbf{GitOps:} Git menyimpan desired state; agen software melakukan reconciliation secara berkala.
            \end{itemize}
        \end{column}
        \begin{column}{0.52\textwidth}
            \centering
            \includegraphics[width=\columnwidth,height=0.50\textheight,keepaspectratio]{deployment-context}
        \end{column}
    \end{columns}
\note{
Sapaan: ``Deployment adalah proses membawa aplikasi dari kode menjadi layanan yang berjalan di klaster. Untuk aplikasi multi-service seperti InvenioRDM, ini berarti mengelola 16+ komponen yang saling bergantung.''

Paradigma: ``Ada dua cara utama: manual, di mana operator menjalankan perintah langsung ke klaster, dan GitOps, di mana Git menjadi single source of truth dan agen software menyelaraskan state klaster secara otomatis.''

Transisi: ``Masalah muncul ketika apa yang berjalan di klaster tidak lagi sesuai dengan apa yang tertulis. Itulah configuration drift.''
}
\end{frame}
}

% =============================================
% 3. Context: Kubernetes and the drift problem
% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Konteks: Kubernetes dan Masalah Drift}
    \begin{columns}[T]
        \begin{column}{0.44\textwidth}
            \scriptsize
            \textbf{Situasi yang sering terjadi}
            \begin{itemize}\setlength{\itemsep}{0pt}
                \item Operator menjalankan \texttt{kubectl scale} langsung di klaster.
                \item Dokumentasi tidak diperbarui.
                \item Tidak ada yang tahu mengapa replika berubah.
            \end{itemize}
            \vspace{0.04cm}
            \textbf{Itulah configuration drift:} \textit{desired state} $\neq$ \textit{actual state}. Dampak: inkonsistensi, pemulihan lambat, audit tidak jelas.
        \end{column}
        \begin{column}{0.52\textwidth}
            \centering
            \includegraphics[width=\columnwidth,height=0.44\textheight,keepaspectratio]{motivation-drift}
        \end{column}
    \end{columns}
    \vspace{0.04cm}
    \slidelead{Drift = klaster menyimpang dari apa yang tertulis. Tim kehilangan kepastian tentang state sistem.}
\note{
Sapaan: ``Bayangkan seorang operator yang harus menangani spike traffic pada pukul 2 pagi. Dia menjalankan kubectl scale langsung di klaster. Masalah terselesaikan, tapi Git tidak tahu. Besok pagi, tidak ada yang bisa menjelaskan mengapa jumlah replika berubah. Itulah configuration drift.''

Definisi: ``Drift adalah celah antara desired state dan actual state. Dan bukti menunjukkan ini bukan kasus langka.''

Transisi: ``Seberapa sering drift ini terjadi? Mari kita lihat buktinya.''
}
\end{frame}
}

% =============================================
% 4. Evidence: drift is not theoretical
% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Bukti: Drift Bukan Masalah Teoritis}
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \scriptsize
            \textbf{Bukti empiris}
            \begin{itemize}\setlength{\itemsep}{1pt}
                \item 71\% dari 2.260 skrip K8s mengandung \textit{misconfiguration} bermakna {\tiny (Zhang, 2026)}.
                \item 11 kategori \textit{security misconfiguration} berulang {\tiny (Rahman, 2023)}.
            \end{itemize}
            \vspace{0.04cm}
            \textbf{Tiga penyebab utama drift} {\tiny (Pohjola, 2025)}
            \begin{enumerate}\setlength{\itemsep}{0pt}
                \item Perubahan langsung tanpa source of truth.
                \item Prosedur deployment tidak konsisten.
                \item Tidak ada mekanisme deteksi otomatis.
            \end{enumerate}
        \end{column}
        \begin{column}{0.47\textwidth}
            \scriptsize
            \textbf{Konsekuensi nyata di klaster akademis}
            \vspace{0.02cm}

            Tim IT universitas yang mengelola repositori data penelitian harus memulihkan layanan setelah drift. Tanpa deteksi otomatis, drift menumpuk sampai insiden.
            \vspace{0.04cm}

            Saat insiden: apa yang berubah, kapan, mengapa? Tanpa audit trail, pemulihan memakan waktu dan akar masalah luput.
        \end{column}
    \end{columns}
    \vspace{0.04cm}
    \slidelead{Drift bukan masalah teoritis; ini ancaman operasional nyata terhadap keandalan dan keamanan layanan.}
\note{
Sapaan: ``Bukti menunjukkan drift bukan masalah teoritis. Zhang et al. menemukan 71\% skrip K8s mengandung misconfiguration bermakna. Rahman et al. mengidentifikasi 11 kategori security misconfiguration yang berulang.''

Konsekuensi: ``Di klaster akademis, konsekuensinya nyata: tim IT universitas harus memulihkan repositori data penelitian setelah drift terjadi. Tanpa mekanisme deteksi otomatis, drift bisa menumpuk sampai insiden.''

Transisi: ``GitOps diajukan sebagai jawaban atas masalah ini. Tapi seberapa besar bukti terkontrol yang kita punya?''
}
\end{frame}
}

% =============================================
% 5. GitOps as response
% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{GitOps Menjanjikan Jawaban}
    \begin{columns}[T]
        \begin{column}{0.42\textwidth}
            \scriptsize
            \textbf{Empat prinsip OpenGitOps}
            \begin{enumerate}\setlength{\itemsep}{1pt}
                \item Sistem dideklarasikan secara deklaratif.
                \item Desired state tersimpan dan berversi di Git.
                \item Perubahan ditarik secara otomatis oleh agen software.
                \item Agen software secara kontinu mendeteksi dan mengoreksi deviasi.
            \end{enumerate}
            \vspace{0.04cm}
            \textbf{Konsekuensi:} drift menjadi \textit{observable} dan dapat dipulihkan otomatis (\textit{self-heal}); audit trail terjamin lewat Git history.
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\columnwidth,height=0.46\textheight,keepaspectratio]{gitops-flow}
        \end{column}
    \end{columns}
    \vspace{0.04cm}
    \smallnote{Pertanyaan ilmiahnya bukan ``apakah GitOps populer?'', melainkan seberapa besar perbedaan operasionalnya saat drift terjadi.}
\note{
Sapaan: ``GitOps menjawab drift melalui empat prinsip OpenGitOps: deklaratif, Git sebagai source of truth, pull-based changes, dan continuous reconciliation. Konsekuensinya: drift menjadi observable dan dapat dipulihkan otomatis.''

Pertanyaan: ``Pertanyaan ilmiahnya bukan apakah GitOps populer, melainkan seberapa besar perbedaan operasionalnya saat drift terjadi.''

Transisi: ``Lalu, seberapa kuat bukti yang mendukung klaim ini?''
}
\end{frame}

\begin{frame}
    \frametitle{Rumusan Masalah}
    \vspace{0.04cm}
    \begin{beamercolorbox}[wd=\textwidth, sep=0.20cm, center, rounded=true]{structure}
        \small\bfseries
        Sejauh mana paradigma deployment manual dan GitOps menghasilkan perbedaan pada karakteristik operasional ketika \textit{configuration drift} terjadi?
    \end{beamercolorbox}
    \vspace{0.10cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \small\textbf{Tiga dimensi pengukuran}
            \begin{enumerate}\setlength{\itemsep}{2pt}
                \item \textbf{R1:} Konsistensi: apakah state klaster kembali ke yang diharapkan setelah drift?
                \item \textbf{R2:} Waktu: berapa lama drift terdeteksi dan dipulihkan?
                \item \textbf{R3:} Identifikasi: berapa lama sumber drift dapat diidentifikasi?
            \end{enumerate}
        \end{column}
        \begin{column}{0.48\textwidth}
            \small\textbf{Skenario pengujian}
            \vspace{0.04cm}
            \scriptsize
            Tujuh skenario drift terkontrol (A--G) yang merepresentasikan kesalahan operasional umum pada Kubernetes: perubahan replika, image, ConfigMap, resource deletion, patch manual, drift lintas namespace, dan perubahan Secret.
            \vspace{0.08cm}
            \small\textbf{Pendekatan analisis}
            \vspace{0.04cm}
            \scriptsize
            Eksperimen sistem terkontrol: mengukur besar perbedaan operasional melalui effect size dan confidence interval, dengan uji statistik sebagai analisis pendukung.
        \end{column}
    \end{columns}
\note{
Sapaan: ``Ini pertanyaan inti penelitian. Bacakan perlahan.'' Lalu: ``Untuk menjawabnya, kita ukur tiga hal: konsistensi, waktu respons, dan waktu identifikasi.''

Skenario: ``Kedua lingkungan diuji terhadap tujuh skenario drift yang merepresentasikan kesalahan operasional umum di Kubernetes.''

Analisis: ``Pendekatan kita eksperimen sistem terkontrol: mengukur seberapa besar perbedaan operasionalnya, dilaporkan melalui effect size dan confidence interval, dengan uji statistik sebagai analisis pendukung.''

Transisi: ``Tapi sebelum kita bahas metodenya, mari kita batasi ruang lingkup agar klaim kita proporsional.''

% 16. Statistical analysis & sample size
% =============================================
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Analisis Statistik dan Ukuran Sampel}
    \vspace{0.02cm}
    \scriptsize
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Hierarki interpretasi}
            \begin{enumerate}\setlength{\itemsep}{1pt}
                \item[\textbf{1.}] \textbf{Effect size} (Cohen's $d$ / Cliff's $\delta$) $\rightarrow$ \textbf{utama}
                \item[\textbf{2.}] \textbf{Confidence interval 95\%} $\rightarrow$ \textbf{utama}
                \item[\textbf{3.}] \textit{p}-value ($t$-test / Mann--Whitney U) $\rightarrow$ pendukung
            \end{enumerate}
            \vspace{0.06cm}
            \textbf{Mengapa effect size di atas?}
            \begin{itemize}\setlength{\itemsep}{1pt}
                \item[$\Rightarrow$] Keputusan deployment bergantung pada \textit{significance praktis}, bukan semata \textit{p}-value.
                \item[$\Rightarrow$] ``Seberapa besar perbedaan operasionalnya?'' bukan ``Apakah perbedaannya signifikan?''
            \end{itemize}
            \vspace{0.04cm}
            \textbf{Uji normalitas:} Shapiro-Wilk.
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Ukuran sampel dan power}
            \begin{itemize}\setlength{\itemsep}{1pt}
                \item Power = 0,80, $\alpha = 0{,}05$, target efek besar ($d = 1{,}0$).
                \item Minimum $n = 17$ per grup secara total.
                \item \textbf{R2:} $7 \times 10 = 70$ observasi per lingkungan (4$\times$ lipat minimum).
                \item \textbf{R1, R3:} $7 \times 5 = 35$ observasi per lingkungan (2$\times$ lipat minimum).
                \item Jika efek sedang teramati, pengulangan ditambah.
            \end{itemize}
            \vspace{0.06cm}
            \begin{beamercolorbox}[wd=\textwidth, sep=0.10cm, rounded=true]{structure}
                \scriptsize\bfseries Kontribusi utama: besar efek dan significance praktis, bukan semata p-value
            \end{beamercolorbox}
        \end{column}
    \end{columns}
\note{
Sapaan: ``Hierarki interpretasi kita: effect size dan confidence interval sebagai utama, p-value sebagai pendukung. Keputusan deployment bergantung pada significance praktis --- seberapa besar perbedaan operasionalnya --- bukan semata apakah p-value di bawah 0,05.''

Power: ``Untuk R2, kita punya 70 observasi per lingkungan, empat kali lipat minimum 17. Untuk R1 dan R3, 35 observasi, dua kali lipat. Jika efek yang teramati ternyata sedang, kita bisa menambah pengulangan.''

Transisi: ``Setelah analisis, kita perlu mengakui ancaman terhadap validitas dan menjelaskan mitigasinya.''
}
\end{frame}

% B2: MTTI definition
% ---------------------------------------------
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Backup: MTTI --- Mean Time To Identify}
    \vspace{0.02cm}
    \scriptsize
    \begin{beamercolorbox}[wd=\textwidth, sep=0.20cm, center, rounded=true]{structure}
        \small\bfseries Pertanyaan penguji: Bagaimana Anda mengukur waktu identifikasi secara objektif?
    \end{beamercolorbox}
    \vspace{0.08cm}
    \textbf{Definisi operasional MTTI:}
    \begin{itemize}
        \item MTTI = $t_{id} - t_d$: waktu dari deteksi drift hingga sumber drift berhasil diidentifikasi.
        \item Identifikasi dianggap selesai ketika operator menyebutkan \textbf{(1)} kategori drift dan \textbf{(2)} sumber daya terdampak, lalu jawaban diverifikasi terhadap skenario injeksi yang diketahui.
    \end{itemize}
    \vspace{0.06cm}
    \textbf{Mengapa MTTI menggantikan rubrik traceability:}
    \begin{itemize}
        \item Menghasilkan \textit{timestamp} objektif, bukan penilaian subjektif.
        \item Langsung mengukur usaha diagnostik operasional.
        \item Terkait langsung dengan \textit{incident response}.
        \item Mudah dijelaskan dan mudah dipertahankan.
    \end{itemize}
    \vspace{0.06cm}
    \textbf{Perbedaan antar-paradigma:}
    \begin{itemize}
        \item Lingkungan B (GitOps): ArgoCD menampilkan \textit{diff}, akar masalah langsung terlihat.
        \item Lingkungan A (Manual): operator memakai \texttt{kubectl describe} dan log, memerlukan pencarian manual.
        \item \textit{Timestamp} dicatat secara otomatis oleh skrip saat jawaban benar disebutkan.
    \end{itemize}
\end{frame}
}

% B3: Why MTTI replaces rubric and manual intervention count
% ---------------------------------------------
{
\usebackgroundtemplate{\backgroundcontent}
\begin{frame}
    \frametitle{Backup: Mengapa MTTI Menggantikan Rubrik dan Intervensi Manual?}
    \vspace{0.02cm}
    \small
    \begin{beamercolorbox}[wd=\textwidth, sep=0.20cm, center, rounded=true]{structure}
        \small\bfseries Pertanyaan penguji: Mengapa tidak mempertahankan rubrik traceability dan hitungan intervensi manual?
    \end{beamercolorbox}
    \vspace{0.08cm}
    \textbf{Intervensi manual (dihapus):}
    \begin{itemize}
        \item Hasilnya sebagian besar sudah ditentukan: manual $> 0$, GitOps $\approx 0$.
        \item Memberikan sedikit informasi baru di luar apa yang sudah diketahui dari desain paradigma.
        \item Menghapusnya membuat desain penelitian lebih bersih dan fokus.
    \end{itemize}
    \vspace{0.06cm}
    \textbf{Rubrik traceability (dihapus):}
    \begin{itemize}
        \item Skala 1--5 rentan terhadap pertanyaan: mengapa skala ini? apakah rubrik benar-benar mengukur traceability?
        \item Cohen's $\kappa$ menambah kompleksitas tanpa menambah kejelasan substantif.
    \end{itemize}
    \vspace{0.06cm}
    \textbf{MTTI menggantikan keduanya:}
    \begin{itemize}
        \item Mengukur aspek yang paling penting dari traceability: \textit{berapa lama} sumber drift bisa diidentifikasi.
        \item Menghasilkan \textit{timestamp} objektif, tidak memerlukan penilaian subjektif.
        \item Terkait langsung dengan \textit{incident response}: MTTI adalah metrik yang digunakan di operasional nyata.
    \end{itemize}
\end{frame}
}
